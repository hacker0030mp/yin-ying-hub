local unlocked = false
local function tryUnlock(code)
    if unlocked then return end
    code = (code or ""):gsub("%s", "")
    if code == "" then setStatus("⚠ ใส่รหัสก่อน") return end
    if Keys[code] or code == MASTER_PASS then
        unlocked = true
        setStatus("✓ ยืนยันสำเร็จ! กำลังปลดล็อค...", Color3.fromRGB(16, 185, 129))
        task.delay(servers 0.9, function()
            TweenService:Create(f, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
                Size = UDim2.new(0, 0, 0, 0),
                Position = UDim2.new(0.5, 0, 0.5, 0)
            }):Play()
            task.wait(0.4)
            sg:Destroy()
        end)
        -- Hook: track success
        _G.DeltaX_Authed = true
    else
        setStatus("✗ รหัสไม่ถูกต้อง")
    end
end
