        function start30sTimer(item) {
            let timeLeft = 30;
            const numEl = document.getElementById('countdown-number');
            const progEl = document.getElementById('countdown-progress');
            numEl.innerText = '30';
            progEl.style.width = '100%';

            if (countdownTimer) clearInterval(countdownTimer);

            countdownTimer = setInterval(() => {
                timeLeft--;
                numEl.innerText = timeLeft;
                progEl.style.width = (timeLeft / 30) * 100 + '%';

                if (timeLeft <= 0) {
                    clearInterval(countdownTimer);
                    document.getElementById('modal-timer-box').classList.add('hidden');
                    document.getElementById('modal-unlocked-box').classList.remove('hidden');
                    document.getElementById('modal-code-display').innerText = item.code;
                    showToast("นับถอยหลังครบ 30 วินาที! ปลดล็อกสคริปต์แล้ว");
                }
            }, 1000);
        }

        async function copyToClipboard(text) {
            try {
                if (navigator.clipboard && window.isSecureContext) await navigator.clipboard.writeText(text);
                else {
                    const ta=document.createElement('textarea'); ta.value=text; ta.style.position='fixed'; ta.style.opacity='0';
                    document.body.appendChild(ta); ta.select(); document.execCommand('copy'); ta.remove();
                }
                showToast("คัดลอกลงคลิปบอร์ดแล้ว!");
            } catch(e) { showToast("คัดลอกไม่สำเร็จ กรุณาเลือกข้อความแล้วคัดลอก"); }
        }
